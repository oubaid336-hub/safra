    // Mobile menu toggle
    const navToggle = document.getElementById('navToggle');
    const mobileMenu = document.getElementById('mobileMenu');
    navToggle.addEventListener('click', () => mobileMenu.classList.toggle('open'));
    mobileMenu.querySelectorAll('a').forEach(a => a.addEventListener('click', () => mobileMenu.classList.remove('open')));

    // Login form
    const form = document.getElementById('loginFormEl');
    const errorBox = document.getElementById('loginError');
    const btn = document.getElementById('loginBtn');

    function showError(message) {
      errorBox.textContent = message;
      errorBox.classList.add('show');
    }
    function hideError() {
      errorBox.classList.remove('show');
      errorBox.textContent = '';
    }

    form.addEventListener('submit', async (e) => {
      e.preventDefault();
      hideError();

      const email = document.getElementById('email').value.trim();
      const password = document.getElementById('password').value;

      if (!email || !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email)) {
        return showError('Please enter a valid email address.');
      }
      if (!password) {
        return showError('Please enter your password.');
      }

      btn.disabled = true;
      btn.textContent = 'Logging in…';

      try {
        const res = await fetch(`${API_BASE_URL}/api/auth/login`, {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          credentials: 'include', // ESSENTIAL: tells the browser to accept and store the httpOnly JWT cookie
          body: JSON.stringify({ email, password })
        });

        const data = await res.json().catch(() => ({}));

        if (!res.ok) {
          // Map common backend error messages to friendly text.
          // The backend's authController distinguishes between "user not found",
          // "wrong password", "not verified", and "account blocked" — we
          // surface each with its own message so the user knows what to do.
          const msg = (data.message || data.error || '').toLowerCase();
          if (res.status === 403 && /blocked/i.test(msg)) {
            showError('This account has been blocked. Please contact support.');
          } else if (/verify/i.test(msg)) {
            showError('Please verify your email first. Check your inbox for the verification link.');
          } else if (/invalid|credentials|password|email/i.test(msg)) {
            showError('Invalid email or password.');
          } else if (msg) {
            showError(data.message || data.error);
          } else {
            showError(`Login failed (${res.status}). Please try again.`);
          }
          btn.disabled = false;
          btn.textContent = 'Log In';
          return;
        }

        // Success — JWT cookie was set by the backend (HttpOnly, Secure, SameSite=Strict).
        // Redirect to homepage. We use location.replace so the login page
        // doesn't stay in browser history (back button won't return to a logged-in
        // session view of the login form).
        window.location.replace('index.html');

      } catch (err) {
        showError('Network error. Check your connection and try again.');
        btn.disabled = false;
        btn.textContent = 'Log In';
      }
    });
  </script>

    // Mobile menu toggle (same as index.html)
    const navToggle = document.getElementById('navToggle');
    const mobileMenu = document.getElementById('mobileMenu');
    navToggle.addEventListener('click', () => mobileMenu.classList.toggle('open'));
    mobileMenu.querySelectorAll('a').forEach(a => a.addEventListener('click', () => mobileMenu.classList.remove('open')));

    // Signup form logic
    const form = document.getElementById('signupFormEl');
    const errorBox = document.getElementById('signupError');
    const btn = document.getElementById('signupBtn');
    const signupForm = document.getElementById('signupForm');
    const signupSuccess = document.getElementById('signupSuccess');
    const successEmail = document.getElementById('successEmail');

    function showError(message) {
      errorBox.textContent = message;
      errorBox.classList.add('show');
    }
    function hideError() {
      errorBox.classList.remove('show');
      errorBox.textContent = '';
    }

    form.addEventListener('submit', async (e) => {
      e.preventDefault();
      hideError();

      const name = document.getElementById('name').value.trim();
      const email = document.getElementById('email').value.trim();
      const password = document.getElementById('password').value;

      // Basic client-side validation before hitting the backend
      if (!name) return showError('Please enter your name.');
      if (!email || !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email)) {
        return showError('Please enter a valid email address.');
      }
      if (password.length < 8) {
        return showError('Password must be at least 8 characters.');
      }

      btn.disabled = true;
      btn.textContent = 'Creating account…';

      try {
        const res = await fetch(`${API_BASE_URL}/api/auth/signup`, {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          credentials: 'include',
          body: JSON.stringify({ name, email, password })
        });

        // Try to parse JSON regardless of status, so we can show the server's error message if any
        const data = await res.json().catch(() => ({}));

        if (!res.ok) {
          // Map common backend error messages to friendly text
          const msg = data.message || data.error || '';
          if (/email.*exists|already.*use|duplicate/i.test(msg)) {
            showError('An account with that email already exists. Try logging in.');
          } else if (msg) {
            showError(msg);
          } else {
            showError(`Signup failed (${res.status}). Please try again.`);
          }
          btn.disabled = false;
          btn.textContent = 'Sign Up';
          return;
        }

        // Success — swap the form for the "check your email" panel
        successEmail.textContent = email;
        signupForm.style.display = 'none';
        signupSuccess.classList.add('show');
        window.scrollTo({ top: 0, behavior: 'smooth' });

      } catch (err) {
        showError('Network error. Check your connection and try again.');
        btn.disabled = false;
        btn.textContent = 'Sign Up';
      }
    });
