// admin/js/bookings.js — D4: Manage bookings (confirm/cancel UI)
//
// UNLIKE D2 (dashboard.js) and D3 (houses.js), Booking.js's exact
// field names were never confirmed against real model/controller
// content in any chat — there's no equivalent to the House.js/Car.js
// field lists this project already has. Everything below that touches
// a Booking field is either a REUSED assumption (already made and not
// contradicted by an earlier, already-built C-phase page) or a NEW
// assumption made here — both kinds are flagged inline. Nothing is
// silently guessed without a comment saying so.
//
// ROUTES (all assumed by following the exact URL pattern CONFIRMED for
// houses/cars — /api/admin/<resource>/:id/<action> — not yet confirmed
// directly against bookingRoutes.js/adminController.js content):
//   GET /api/admin/bookings
//   PUT /api/admin/bookings/:id/confirm
//   PUT /api/admin/bookings/:id/cancel
// A13 (Phase A) confirmed admin confirm/cancel WORKS live against some
// route — just not necessarily this exact string. Test this page's
// buttons for real before trusting the guess.
//
// FIELDS:
//   status ("pending" | "confirmed" | "cancelled") and totalPrice —
//   CONFIRMED, bookingController.js content was already read for these
//   (see Known Issue #11 and the C5 build note in the master plan).
//   listingId + listingType ("house" | "car") — REUSED assumption,
//   same one C7 (wishlist.js), C8 (my-bookings.js), and C9 (reviews.js)
//   already made. Used here to fetch the listing's title the same way
//   C8 does: GET /api/houses/:id or /api/cars/:id per booking.
//   Guest identity — NEW assumption: tries a populated `user` object
//   (name/email) first, then falls back to a raw id field (userId,
//   customerId, or user-as-string), labeled explicitly as an ID rather
//   than left blank or invented.
//   Stay dates — NEW assumption: tries common field-name pairs in
//   order (checkIn/checkOut, startDate/endDate, dateFrom/dateTo). If
//   none of those exist on the record, the card just omits the dates
//   line instead of fabricating one.

(function () {
  const statusBanner = document.getElementById('statusBanner');
  const loadingMsg = document.getElementById('loadingMsg');
  const emptyMsg = document.getElementById('emptyMsg');
  const bookingsList = document.getElementById('bookingsList');
  const logoutBtn = document.getElementById('logoutBtn');

  function showError(text) {
    statusBanner.textContent = text;
    statusBanner.classList.add('show');
  }

  function clearError() {
    statusBanner.classList.remove('show');
  }

  // Pending first (needs action), then confirmed, then cancelled.
  function sortByStatus(items) {
    const order = { pending: 0, confirmed: 1, cancelled: 2 };
    return [...items].sort((a, b) => (order[a.status] ?? 3) - (order[b.status] ?? 3));
  }

  function formatDate(value) {
    if (!value) return null;
    const d = new Date(value);
    if (isNaN(d.getTime())) return null;
    return d.toLocaleDateString(undefined, { year: 'numeric', month: 'short', day: 'numeric' });
  }

  // Guest identity: populated object first, raw id fallback second.
  function getGuestLabel(booking) {
    const candidate = booking.user || booking.guest || booking.customer;
    if (candidate && typeof candidate === 'object') {
      if (candidate.name && candidate.email) return `${candidate.name} (${candidate.email})`;
      if (candidate.name) return candidate.name;
      if (candidate.email) return candidate.email;
    }
    const rawId = booking.userId || booking.customerId ||
      (typeof booking.user === 'string' ? booking.user : null);
    if (rawId) return `Guest ID: ${rawId}`;
    return 'Guest info unavailable';
  }

  // Tries known date field-name pairs in order; first pair where BOTH
  // sides parse to a valid date wins.
  function getDatesLabel(booking) {
    const pairs = [
      ['checkIn', 'checkOut'],
      ['startDate', 'endDate'],
      ['dateFrom', 'dateTo']
    ];
    for (const [fromKey, toKey] of pairs) {
      const from = formatDate(booking[fromKey]);
      const to = formatDate(booking[toKey]);
      if (from && to) return `${from} → ${to}`;
    }
    return null;
  }

  function getListingType(booking) {
    const type = booking.listingType;
    if (type === 'house' || type === 'car') return type;
    return null;
  }

  function getListingId(booking) {
    return booking.listingId || booking.listing || null;
  }

  // Same lookup C8 (my-bookings.js) already does: fetch the listing's
  // own title separately, since it's assumed NOT pre-populated on the
  // booking response. Fails gracefully rather than breaking the card.
  async function fetchListingTitle(booking) {
    const type = getListingType(booking);
    const id = getListingId(booking);
    if (!type || !id) return 'Listing unavailable';

    const endpoint = type === 'house' ? 'houses' : 'cars';
    try {
      const res = await fetch(`${API_BASE_URL}/api/${endpoint}/${id}`, {
        method: 'GET',
        credentials: 'include'
      });
      if (!res.ok) return 'Listing unavailable';
      const data = await res.json();
      return data.title || 'Listing unavailable';
    } catch (err) {
      console.error(`Failed to fetch ${type} ${id} for booking:`, err);
      return 'Listing unavailable';
    }
  }

  function buildCard(booking, listingTitle) {
    const card = document.createElement('div');
    card.className = 'booking-card';

    const info = document.createElement('div');
    info.className = 'booking-info';

    const type = getListingType(booking);
    const title = document.createElement('div');
    title.className = 'title';
    title.textContent = listingTitle;
    if (type) {
      const typeTag = document.createElement('span');
      typeTag.className = 'type-tag';
      typeTag.textContent = type;
      title.appendChild(typeTag);
    }

    const guestMeta = document.createElement('div');
    guestMeta.className = 'meta';
    const price = booking.totalPrice != null ? `$${booking.totalPrice}` : 'price n/a';
    const dates = getDatesLabel(booking);
    guestMeta.textContent = [getGuestLabel(booking), dates, price].filter(Boolean).join(' · ');

    info.appendChild(title);
    info.appendChild(guestMeta);

    const bookedOnDate = formatDate(booking.createdAt);
    if (bookedOnDate) {
      const bookedOn = document.createElement('div');
      bookedOn.className = 'booked-on';
      bookedOn.textContent = `Booked on ${bookedOnDate}`;
      info.appendChild(bookedOn);
    }

    const badge = document.createElement('span');
    badge.className = 'status-badge ' + (booking.status || 'pending');
    badge.textContent = booking.status || 'pending';

    const actions = document.createElement('div');
    actions.className = 'booking-actions';

    const confirmBtn = document.createElement('button');
    confirmBtn.className = 'confirm-btn';
    confirmBtn.textContent = 'Confirm';
    confirmBtn.disabled = booking.status === 'confirmed' || booking.status === 'cancelled';

    const cancelBtn = document.createElement('button');
    cancelBtn.className = 'cancel-btn';
    cancelBtn.textContent = 'Cancel';
    cancelBtn.disabled = booking.status === 'cancelled';

    confirmBtn.addEventListener('click', () => setBookingStatus(booking._id, 'confirm', card));
    cancelBtn.addEventListener('click', () => setBookingStatus(booking._id, 'cancel', card));

    actions.appendChild(confirmBtn);
    actions.appendChild(cancelBtn);

    card.appendChild(info);
    card.appendChild(badge);
    card.appendChild(actions);

    return card;
  }

  async function setBookingStatus(id, action, cardEl) {
    const buttons = cardEl.querySelectorAll('button');
    buttons.forEach((b) => (b.disabled = true));
    clearError();

    try {
      const res = await fetch(`${API_BASE_URL}/api/admin/bookings/${id}/${action}`, {
        method: 'PUT',
        credentials: 'include'
      });

      if (res.status === 401 || res.status === 403) {
        window.location.href = 'login.html';
        return;
      }

      if (!res.ok) {
        showError('Something went wrong updating that booking. Please try again.');
        buttons.forEach((b) => (b.disabled = false));
        return;
      }

      // Reload fresh rather than patch local state — keeps this in
      // sync with whatever the server actually did (same call D3 makes).
      await loadAll();

    } catch (err) {
      console.error(`Failed to ${action} booking ${id}:`, err);
      showError('Something went wrong updating that booking. Please try again.');
      buttons.forEach((b) => (b.disabled = false));

// This same fetch call also acts as the page's auth gate: a 401/403

(function () {
  const statusBanner = document.getElementById('statusBanner');
  const loadingMsg = document.getElementById('loadingMsg');
  const emptyMsg = document.getElementById('emptyMsg');

  function formatLabel(key) {
    const withSpaces = key.replace(/([a-z0-9])([A-Z])/g, '$1 $2');
    return withSpaces
      .replace(/_/g, ' ')
      .replace(/\b\w/g, (c) => c.toUpperCase());
  }

  function isPlainObject(value) {
    return value !== null && typeof value === 'object' && !Array.isArray(value);
  }

    statsGrid.innerHTML = '';
    const keys = Object.keys(data || {});

    if (keys.length === 0) {
      emptyMsg.style.display = 'block';

        const groupCard = document.createElement('div');
        groupCard.className = 'stat-card nested';

        const groupLabel = document.createElement('div');
        groupLabel.className = 'stat-label group-label';

          subWrap.appendChild(subValue);
          subWrap.appendChild(subLabel);
          groupCard.appendChild(subWrap);
        });

        return;

        const card = document.createElement('div');

        valEl.textContent = value.length;

        labelEl.textContent = formatLabel(key) + ' (list)';

      card.className = 'stat-card';
      const valEl = document.createElement('div');

      const labelEl = document.createElement('div');

// admin/js/login.js - D1: Admin login page
//
// Design note: this reuses the SAME POST /api/auth/login endpoint the
// regular customer login.html already calls (per the plan: "reuses Part
// 4 auth"). There is no separate admin-only login route.
//
// We do NOT trust the login response body to tell us whether this user
// is an admin - its exact shape was never confirmed in any chat, so
// guessing at a field name there would be exactly the kind of unverified
// assumption this project tries to avoid. Instead, right after login
// succeeds (cookie is set), we call the CONFIRMED admin-only route
// GET /api/admin/dashboard. If that 200s, they're really an admin and we
// redirect in. If it 403s, they're logged in as themselves but not an
// admin - same as the reminder in the plan: the page is just a friendly
// gate, the real security boundary is requireAdmin on the backend.

(function () {
  const form = document.getElementById('adminLoginForm');
  const emailInput = document.getElementById('email');
  const passwordInput = document.getElementById('password');
  const submitBtn = document.getElementById('submitBtn');
  const banner = document.getElementById('formBanner');

  function showError(text) {
    banner.textContent = text;
    banner.className = 'form-banner show error';
  }

  function clearBanner() {
    banner.className = 'form-banner';
  }

  form.addEventListener('submit', async function (e) {
    e.preventDefault();
    clearBanner();

    const email = emailInput.value.trim();
    const password = passwordInput.value;

    if (!email || !password) {
      showError('Please enter your email and password.');
      return;
    }

    submitBtn.disabled = true;
    submitBtn.textContent = 'Checking...';

    try {
      const loginRes = await fetch(`${API_BASE_URL}/api/auth/login`, {
        method: 'POST',
        credentials: 'include',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ email, password })
      });

      if (!loginRes.ok) {
        showError('Invalid email or password.');
        return;
      }

      const dashboardRes = await fetch(`${API_BASE_URL}/api/admin/dashboard`, {
        method: 'GET',
        credentials: 'include'
      });

      if (dashboardRes.status === 403) {
        showError("This account doesn't have admin access.");
        return;
      }

      if (!dashboardRes.ok) {
        showError('Something went wrong. Please try again.');
        return;
      }

      window.location.href = 'dashboard.html';

    } catch (err) {
      console.error('Admin login failed:', err);
      showError('Something went wrong. Please try again.');
    } finally {
      submitBtn.disabled = false;
      submitBtn.textContent = 'Log In';
    }
  });
})();

// admin/js/users.js — D5: Manage users (block/unblock/delete UI)
//
// FIELDS: "role" ("customer"/"admin") and "verified" are CONFIRMED
// real User fields — role is discussed directly in Known Issue #10,
// and verified is the exact field D1/booking's 403 flow already
// depends on. "name" and blocked-status are NOT confirmed the same
// way — no User.js field list like House.js/Car.js ever got pasted
// into a chat. Handled the same defensive way as D4:
//   - Display name tries user.name first, falls back to user.email
//     alone (guaranteed to exist — every login/signup flow in this
//     project is email-based).
//   - Blocked-status badge tries common candidate fields (isBlocked,
//     blocked, or status === 'blocked') and falls back to treating
//     the user as active if none of those are present, rather than
//     guessing wrong in the alarming direction.
// This guessing only affects the BADGE — the Block/Unblock buttons
// call the same routes either way and don't depend on knowing which
// field the backend actually uses.
//
// ROUTES (assumed by the same /api/admin/<resource>/:id/<action>
// pattern used for D3/D4, not confirmed directly against
// adminController.js/userRoutes.js content — A7-A9 confirmed
// block/unblock/delete WORK live, just not necessarily these exact
// strings):
//   GET /api/admin/users
//   PUT /api/admin/users/:id/block
//   PUT /api/admin/users/:id/unblock
//   DELETE /api/admin/users/:id
// NOTE: unlike D3's delete (confirmed on the public /api/houses/:id
// path), there's no evidence users have an equivalent public route —
// so DELETE here is assumed under /api/admin/, not following D3's
// exception.
//
// SELF-PROTECTION: fetches the logged-in admin's own id via the
// CONFIRMED GET /api/auth/me and disables Block/Delete on their own
// row, so an admin can't accidentally lock themselves out or delete
// their own account from this page.
//
// KNOWN ISSUE #16 REMINDER: blocking/deleting here has no effect on
// an already-active session until it expires or the user logs out
// (requireAuth never re-checks the DB). Surfaced in the page's
// scope-note, not repeated per-row to avoid clutter.

(function () {
  const statusBanner = document.getElementById('statusBanner');
  const loadingMsg = document.getElementById('loadingMsg');
  const emptyMsg = document.getElementById('emptyMsg');
  const usersList = document.getElementById('usersList');
  const logoutBtn = document.getElementById('logoutBtn');

  let currentAdminId = null;

  function showError(text) {
    statusBanner.textContent = text;
    statusBanner.classList.add('show');
  }

  function clearError() {
    statusBanner.classList.remove('show');
  }

  function isUserBlocked(user) {
    if (typeof user.isBlocked === 'boolean') return user.isBlocked;
    if (typeof user.blocked === 'boolean') return user.blocked;
    if (typeof user.status === 'string') return user.status === 'blocked';
    return false; // unknown shape — default to "active" rather than alarm
  }

  function getDisplayName(user) {
    return user.name || user.email || 'Unknown user';
  }

  // Admins first, then customers, alphabetically within each group —
  // easier to scan than server order when the list grows.
  function sortUsers(users) {
    const roleOrder = { admin: 0, customer: 1 };
    return [...users].sort((a, b) => {
      const roleDiff = (roleOrder[a.role] ?? 2) - (roleOrder[b.role] ?? 2);
      if (roleDiff !== 0) return roleDiff;
      return getDisplayName(a).localeCompare(getDisplayName(b));
    });
  }

  function buildCard(user) {
    const card = document.createElement('div');
    card.className = 'user-card';

    const info = document.createElement('div');
    info.className = 'user-info';

    const nameLine = document.createElement('div');
    nameLine.className = 'name-line';
    nameLine.textContent = getDisplayName(user);

    const meta = document.createElement('div');
    meta.className = 'meta';
    // Only show email separately if a real name was available above —
    // otherwise the name line already IS the email.
    meta.textContent = user.name && user.email ? user.email : '';

    info.appendChild(nameLine);
    if (meta.textContent) info.appendChild(meta);

    const tagRow = document.createElement('div');
    tagRow.className = 'tag-row';

    const isSelf = currentAdminId && user._id === currentAdminId;
    const blocked = isUserBlocked(user);

    const roleTag = document.createElement('span');
    roleTag.className = 'tag ' + (user.role === 'admin' ? 'role-admin' : 'role-customer');
    roleTag.textContent = user.role || 'customer';
    tagRow.appendChild(roleTag);

    const statusTag = document.createElement('span');
    statusTag.className = 'tag ' + (blocked ? 'status-blocked' : 'status-active');
    statusTag.textContent = blocked ? 'blocked' : 'active';
    tagRow.appendChild(statusTag);

    if (user.verified === false) {
      const unverifiedTag = document.createElement('span');
      unverifiedTag.className = 'tag unverified';
      unverifiedTag.textContent = 'unverified';
      tagRow.appendChild(unverifiedTag);
    }

    if (isSelf) {
      const selfTag = document.createElement('span');
      selfTag.className = 'tag self-tag';
      selfTag.textContent = 'you';
      tagRow.appendChild(selfTag);
    }

    const actions = document.createElement('div');
    actions.className = 'user-actions';

    const toggleBtn = document.createElement('button');
    toggleBtn.className = blocked ? 'unblock-btn' : 'block-btn';
    toggleBtn.textContent = blocked ? 'Unblock' : 'Block';
    toggleBtn.disabled = isSelf;

    const deleteBtn = document.createElement('button');
    deleteBtn.className = 'delete-btn';
    deleteBtn.textContent = 'Delete';
    deleteBtn.disabled = isSelf;

    toggleBtn.addEventListener('click', () =>
      setBlockedState(user._id, blocked ? 'unblock' : 'block', card)
    );
    deleteBtn.addEventListener('click', () =>
      deleteUser(user._id, getDisplayName(user), card)
    );

    actions.appendChild(toggleBtn);
    actions.appendChild(deleteBtn);

    card.appendChild(info);
    card.appendChild(tagRow);
    card.appendChild(actions);

    return card;
  }

  async function setBlockedState(id, action, cardEl) {
    const buttons = cardEl.querySelectorAll('button');
    buttons.forEach((b) => (b.disabled = true));
    clearError();

    try {
      const res = await fetch(`${API_BASE_URL}/api/admin/users/${id}/${action}`, {
        method: 'PUT',
        credentials: 'include'
      });

      if (res.status === 401 || res.status === 403) {
        window.location.href = 'login.html';
        return;
      }

      if (!res.ok) {
        showError('Something went wrong updating that user. Please try again.');
        buttons.forEach((b) => (b.disabled = false));
        return;
      }

      await loadAll();

    } catch (err) {
      console.error(`Failed to ${action} user ${id}:`, err);
      showError('Something went wrong updating that user. Please try again.');
      buttons.forEach((b) => (b.disabled = false));
    }
  }

  async function deleteUser(id, name, cardEl) {
    const confirmed = window.confirm(
      `Delete "${name}" permanently? This can't be undone.`
    );
    if (!confirmed) return;

    const buttons = cardEl.querySelectorAll('button');
    buttons.forEach((b) => (b.disabled = true));
    clearError();

    try {
      const res = await fetch(`${API_BASE_URL}/api/admin/users/${id}`, {
        method: 'DELETE',
        credentials: 'include'
      });

      if (res.status === 401 || res.status === 403) {
        window.location.href = 'login.html';
        return;
      }

      if (!res.ok) {
        showError('Something went wrong deleting that user. Please try again.');
        buttons.forEach((b) => (b.disabled = false));
        return;
      }

      await loadAll();

    } catch (err) {
      console.error(`Failed to delete user ${id}:`, err);
      showError('Something went wrong deleting that user. Please try again.');
      buttons.forEach((b) => (b.disabled = false));
    }
  }

  // Confirmed route (added to authController.js/authRoutes.js during
  // Phase D prep) — used only to identify "this is me" for the
  // self-protection guard, not as an auth gate (loadAll's own fetch
  // already handles 401/403).
  async function loadCurrentAdminId() {
    try {
      const res = await fetch(`${API_BASE_URL}/api/auth/me`, {
        method: 'GET',
        credentials: 'include'
      });
      if (!res.ok) return null;
      const data = await res.json();
      return data._id || data.id || null;
    } catch (err) {
      console.error('Failed to load current admin identity:', err);
      return null;
    }
  }

  async function loadAll() {
    try {
      const [me, res] = await Promise.all([
        currentAdminId ? Promise.resolve(currentAdminId) : loadCurrentAdminId(),
        fetch(`${API_BASE_URL}/api/admin/users`, { method: 'GET', credentials: 'include' })
      ]);
      currentAdminId = me;

      if (res.status === 401 || res.status === 403) {
        window.location.href = 'login.html';
        return;
      }

      if (!res.ok) {
        loadingMsg.style.display = 'none';
        showError('Something went wrong loading users. Please try again.');
        return;
      }

      const users = await res.json();
      loadingMsg.style.display = 'none';
      usersList.innerHTML = '';

      if (!users || users.length === 0) {
        emptyMsg.style.display = 'block';
        return;
      }
      emptyMsg.style.display = 'none';

      sortUsers(users).forEach((user) => {
        usersList.appendChild(buildCard(user));
      });

    } catch (err) {
      console.error('Failed to load users:', err);
      loadingMsg.style.display = 'none';
      showError('Something went wrong loading users. Please try again.');
    }
  }

  logoutBtn.addEventListener('click', async function () {
    try {
      await fetch(`${API_BASE_URL}/api/auth/logout`, {
        method: 'POST',
        credentials: 'include'
      });
    } catch (err) {
      console.error('Logout request failed:', err);
    } finally {
      window.location.href = 'login.html';
    }
  });

  loadAll();
})();
