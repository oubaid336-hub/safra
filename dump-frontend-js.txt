// booking.js — C5: Booking form logic
//
// ASSUMPTIONS (flag these against the real backend once confirmed):
//   1. This page is opened as booking.html?houseId=<id>
//      -> the "Book Now" link/button on the house detail page (C4)
//         needs to point here with that exact param name.
//   2. GET  ${API_BASE_URL}/api/houses/:id   returns a single house
//      object: { _id, title, description, location, pricePerNight,
//      photos: [] }. This route was never explicitly confirmed against
//      houseController.js/houseRoutes.js (still unread) — only the
//      public LIST endpoint (GET /api/houses) was confirmed live in
//      Phase A. If the real route/shape differs, adjust loadHouse()
//      below.
//   3. POST ${API_BASE_URL}/api/bookings expects JSON body:
//      { listingId, listingType: "house", startDate, endDate }
//      and relies on the httpOnly JWT cookie for auth (credentials:
//      'include'), matching the auth pattern already confirmed in
//      Phase A. totalPrice is NOT sent — the server calculates it
//      (confirmed pattern: $80/night x 4 nights = $320 in Phase A4).

(function () {
  const params = new URLSearchParams(window.location.search);
  const houseId = params.get('houseId');

  const loadingState = document.getElementById('loading-state');
  const notFoundState = document.getElementById('not-found-state');
  const layout = document.getElementById('booking-layout');

  const photoEl = document.getElementById('house-photo');
  const titleEl = document.getElementById('house-title');
  const locationEl = document.getElementById('house-location');
  const descriptionEl = document.getElementById('house-description');

  const startDateInput = document.getElementById('start-date');
  const endDateInput = document.getElementById('end-date');
  const priceBreakdown = document.getElementById('price-breakdown');
  const pricePerNightValue = document.getElementById('price-per-night-value');
  const nightsValue = document.getElementById('nights-value');
  const totalValue = document.getElementById('total-value');

  const form = document.getElementById('booking-form');
  const submitBtn = document.getElementById('submit-btn');
  const formMessage = document.getElementById('form-message');

  let currentHouse = null;

  // No houseId in the URL at all -> nothing to book, bail early.
  if (!houseId) {
    showNotFound();
    return;
  }

  // Prevent picking a check-in date in the past.
  const today = new Date().toISOString().split('T')[0];
  startDateInput.min = today;
  endDateInput.min = today;

  loadHouse(houseId);

  startDateInput.addEventListener('change', handleDatesChanged);
  endDateInput.addEventListener('change', handleDatesChanged);
  form.addEventListener('submit', handleSubmit);

  async function loadHouse(id) {
    try {
      const res = await fetch(`${API_BASE_URL}/api/houses/${id}`, {
        credentials: 'include',
      });

      if (!res.ok) {
        showNotFound();
        return;
      }

      const data = await res.json();
      // Some APIs wrap the object as { house: {...} } — handle both shapes.
      currentHouse = data.house || data;

      renderHouse(currentHouse);
      loadingState.style.display = 'none';
      layout.style.display = 'grid';
    } catch (err) {
      console.error('Failed to load house:', err);
      showNotFound();
    }
  }

  function renderHouse(house) {
    titleEl.textContent = house.title || 'Untitled listing';
    locationEl.textContent = house.location || '';
    descriptionEl.textContent = house.description || '';

    // House.photos is currently an empty array for every listing
    // (Cloudinary upload not wired up yet) — fall back to a plain
    // placeholder rather than a broken <img>.
    if (house.photos && house.photos.length > 0) {
      photoEl.src = house.photos[0];
      photoEl.alt = house.title || 'House photo';
    } else {
      photoEl.removeAttribute('src');
      photoEl.alt = 'No photo available yet';
    }
  }

  function handleDatesChanged() {
    const start = startDateInput.value;
    const end = endDateInput.value;

    if (!start || !end || !currentHouse) {
      priceBreakdown.style.display = 'none';
      return;
    }

    const nights = nightsBetween(start, end);

    if (nights <= 0) {
      priceBreakdown.style.display = 'none';
      showMessage('Check-out must be after check-in.', 'warning');
      return;
    }

    clearMessage();

    const pricePerNight = Number(currentHouse.pricePerNight) || 0;
    const total = nights * pricePerNight;

    pricePerNightValue.textContent = formatPrice(pricePerNight);
    nightsValue.textContent = nights;
    totalValue.textContent = formatPrice(total);
    priceBreakdown.style.display = 'block';
  }

  function nightsBetween(startStr, endStr) {
    const start = new Date(startStr);
    const end = new Date(endStr);
    const msPerNight = 1000 * 60 * 60 * 24;
    return Math.round((end - start) / msPerNight);
  }

  function formatPrice(amount) {
    return `$${amount.toFixed(2)}`;
  }

  async function handleSubmit(e) {
    e.preventDefault();
    clearMessage();

    const startDate = startDateInput.value;
    const endDate = endDateInput.value;

    if (!startDate || !endDate || nightsBetween(startDate, endDate) <= 0) {
      showMessage('Please choose a valid check-in and check-out date.', 'error');
      return;
    }

    submitBtn.disabled = true;
    submitBtn.textContent = 'Submitting…';

    try {
      const res = await fetch(`${API_BASE_URL}/api/bookings`, {
        method: 'POST',
        credentials: 'include',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          listingId: houseId,
          listingType: 'house',
          startDate,
          endDate,
        }),
      });

      if (res.status === 401) {
        // Not logged in — send to login, then back here afterwards.
        const redirectTarget = encodeURIComponent(window.location.pathname + window.location.search);
        window.location.href = `login.html?redirect=${redirectTarget}`;
        return;
      }

      if (res.status === 403) {
        // Known Issue #13: this is the path reserved for
        // "please verify your email before booking". As of this
        // writing, bookingController.js does NOT actually send this
        // check yet (still unread/unwritten) — so this branch is
        // ready on the frontend but nothing on the backend triggers
        // it yet. Once that patch lands, this message will surface
        // correctly without any frontend change needed.
        const data = await safeJson(res);
        showMessage(
          data.message || 'Please verify your email before booking. Check your inbox for the verification link.',
          'warning'
        );
        return;
      }

      if (!res.ok) {
        const data = await safeJson(res);
        showMessage(data.message || 'Something went wrong. Please try again.', 'error');
        return;
      }

      const booking = await safeJson(res);
      showMessage(
        'Booking requested! A confirmation email is on its way — check your inbox.',
        'success'
      );
      form.reset();
      priceBreakdown.style.display = 'none';
    } catch (err) {
      console.error('Booking submission failed:', err);
      showMessage('Could not reach the server. Please check your connection and try again.', 'error');
    } finally {
      submitBtn.disabled = false;
      submitBtn.textContent = 'Request Booking';
    }
  }

  async function safeJson(res) {
    try {
      return await res.json();
    } catch {
      return {};
    }
  }

  function showMessage(text, type) {
    formMessage.textContent = text;
    formMessage.className = `form-message ${type}`;
  }

  function clearMessage() {
    formMessage.textContent = '';
    formMessage.className = 'form-message';
  }

  function showNotFound() {
    loadingState.style.display = 'none';
    layout.style.display = 'none';
    notFoundState.style.display = 'block';
  }
})();

let allCars = [];

function renderCars(list) {
  const grid = document.getElementById('carGrid');

  if (!list.length) {
    grid.innerHTML = '<p class="listing-empty">No cars match your search.</p>';
    return;
  }

  grid.innerHTML = list.map(car => `
    <a class="listing-card" href="car-detail.html?carId=${car._id}">
      <div class="listing-thumb">
        ${car.photos && car.photos[0]
          ? `<img src="${car.photos[0]}" alt="${car.title}" style="width:100%;height:100%;object-fit:cover;" loading="lazy">`
          : `<svg viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="1.5"><path d="M5 17h14M5 17a2 2 0 1 1-4 0 2 2 0 0 1 4 0zm14 0a2 2 0 1 0 4 0 2 2 0 0 0-4 0zM3 17V9l2-5h14l2 5v8"/></svg>`}
      </div>
      <div class="listing-body">
        <div class="listing-title">${car.title}</div>
        <div class="listing-location">
          <svg viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2"><path d="M21 10c0 7-9 12-9 12s-9-5-9-12a9 9 0 0 1 18 0z"/><circle cx="12" cy="10" r="3"/></svg>
          <span>${car.location}</span>
        </div>
        <div class="listing-footer-row">
          <div class="listing-price">$${car.pricePerDay}<span> / day</span></div>
          <span class="listing-cta">View details -&gt;</span>
        </div>
      </div>
    </a>
  `).join('');
}

function applyFilters() {
  const loc = document.getElementById('filterLocation').value.trim().toLowerCase();
  const transmission = document.getElementById('filterTransmission').value;
  const maxPrice = parseFloat(document.getElementById('filterMaxPrice').value);

  const filtered = allCars.filter(c => {
    const matchesLocation = !loc || (c.location || '').toLowerCase().includes(loc);
    const matchesTransmission = !transmission || c.transmission === transmission;
    const matchesMaxPrice = isNaN(maxPrice) || c.pricePerDay <= maxPrice;
    return matchesLocation && matchesTransmission && matchesMaxPrice;
  });

  renderCars(filtered);
}

async function loadCars() {
  const grid = document.getElementById('carGrid');
  try {
    const res = await fetch(`${API_BASE_URL}/api/cars`);
    if (!res.ok) throw new Error(`Server responded with ${res.status}`);
    allCars = await res.json();

    if (!allCars.length) {
      grid.innerHTML = '<p class="listing-empty">No cars available right now.</p>';
      return;
    }

    renderCars(allCars);

    document.getElementById('filterLocation').addEventListener('input', applyFilters);
    document.getElementById('filterTransmission').addEventListener('change', applyFilters);
    document.getElementById('filterMaxPrice').addEventListener('input', applyFilters);
  } catch (err) {
    console.error('Failed to load cars:', err);
    grid.innerHTML = '<p class="listing-error">Could not load cars right now.</p>';
  }
}

loadCars();

// contact.js - handles the public Contact form (C11)
//
// ASSUMPTION (flagged per project convention): POSTs to
// `${API_BASE_URL}/api/contact`. contactController.js / contactRoutes.js
// content has never been read in this project, so this path is inferred
// from the same naming pattern used elsewhere (e.g. /api/faq), not
// confirmed. If it turns out to be something else, only the URL string
// below needs to change - nothing else in this file depends on it.
//
// No login is required for this form (ContactMessage.js has no userId
// field), so this does NOT send credentials: 'include'.

(function () {
  const nameInput = document.getElementById('name');
  const emailInput = document.getElementById('email');
  const messageInput = document.getElementById('message');
  const form = document.getElementById('contactForm');
  const submitBtn = document.getElementById('submitBtn');
  const banner = document.getElementById('formBanner');

  const nameError = document.getElementById('nameError');
  const emailError = document.getElementById('emailError');
  const messageError = document.getElementById('messageError');

  const navToggle = document.getElementById('navToggle');
  const navLinks = document.getElementById('navLinks');
  if (navToggle && navLinks) {
    navToggle.addEventListener('click', function () {
      const isOpen = navLinks.classList.toggle('open');
      navToggle.setAttribute('aria-expanded', isOpen ? 'true' : 'false');
    });
  }

  function showBanner(type, text) {
    banner.textContent = text;
    banner.className = 'form-banner show ' + type;
  }

  function clearFieldErrors() {
    nameError.textContent = '';
    emailError.textContent = '';
    messageError.textContent = '';
  }

  function validate() {
    clearFieldErrors();
    let valid = true;

    const name = nameInput.value.trim();
    const email = emailInput.value.trim();
    const message = messageInput.value.trim();

    if (!name) {
      nameError.textContent = 'Please enter your name.';
      valid = false;
    }

    const emailPattern = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
    if (!email) {
      emailError.textContent = 'Please enter your email.';
      valid = false;
    } else if (!emailPattern.test(email)) {
      emailError.textContent = 'Please enter a valid email address.';
      valid = false;
    }

    if (!message) {
      messageError.textContent = 'Please enter a message.';
      valid = false;
    } else if (message.length < 10) {
      messageError.textContent = 'Message should be at least 10 characters.';
      valid = false;
    }

    return valid ? { name, email, message } : null;
  }

  form.addEventListener('submit', async function (e) {
    e.preventDefault();
    banner.className = 'form-banner';

    const data = validate();
    if (!data) return;

    submitBtn.disabled = true;
    submitBtn.textContent = 'Sending...';

    try {
      const res = await fetch(`${API_BASE_URL}/api/contact`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(data)
      });

      if (res.ok) {
        showBanner('success', "Thanks - your message has been sent. We'll get back to you soon.");
        form.reset();
      } else if (res.status === 400) {
        showBanner('error', 'Please check your details and try again.');
      } else {
        showBanner('error', 'Something went wrong sending your message. Please try again in a moment.');
      }
    } catch (err) {
      console.error('Contact form submit failed:', err);
      showBanner('error', 'Something went wrong sending your message. Please try again in a moment.');
    } finally {
      submitBtn.disabled = false;
      submitBtn.textContent = 'Send Message';
    }
  });
})();

// ASSUMPTION: GET ${API_BASE_URL}/api/faq is the public list endpoint.

function applyFilters() {
  const loc = document.getElementById('filterLocation').value.trim().toLowerCase();
  const minGuests = parseFloat(document.getElementById('filterMinGuests').value);
  const maxPrice = parseFloat(document.getElementById('filterMaxPrice').value);

  const filtered = allHouses.filter(h => {
    const matchesLocation = !loc || (h.location || '').toLowerCase().includes(loc);
    const matchesMinGuests = isNaN(minGuests) || (h.capacity || 0) >= minGuests;
    const matchesMaxPrice = isNaN(maxPrice) || h.pricePerNight <= maxPrice;
    return matchesLocation && matchesMinGuests && matchesMaxPrice;
  });

  renderHouses(filtered);
}

(function () {
  const list = document.getElementById('booking-list');
  const loadingState = document.getElementById('loading-state');
  const loginRequiredState = document.getElementById('login-required-state');
  const emptyState = document.getElementById('empty-state');

  checkAuthThenLoad();

  async function checkAuthThenLoad() {
    try {
      const meRes = await fetch(`${API_BASE_URL}/api/auth/me`, {
        credentials: 'include',
      });

      if (!meRes.ok) {
        showLoginRequired();
        return;
      }

      loadBookings();
    } catch (err) {
      console.error('Auth check failed:', err);
      showLoginRequired();
    }
  }

  async function loadBookings() {
    try {
      const res = await fetch(`${API_BASE_URL}/api/bookings/mine`, {
        credentials: 'include',
      });

      if (res.status === 401) {
        showLoginRequired();
        return;
      }

      if (!res.ok) {
        loadingState.textContent = 'Something went wrong loading your bookings. Please try again later.';
        return;
      }

      const data = await res.json();
      let bookings = data.bookings || data || [];

      // Most recent first, if createdAt is present.
      bookings = [...bookings].sort((a, b) => {
        if (!a.createdAt || !b.createdAt) return 0;
        return new Date(b.createdAt) - new Date(a.createdAt);
      });

      loadingState.style.display = 'none';

      if (!bookings.length) {
        emptyState.style.display = 'block';
        return;
      }

      for (const booking of bookings) {
        renderBooking(booking);
      }
    } catch (err) {
      console.error('Failed to load bookings:', err);
      loadingState.textContent = 'Could not reach the server. Please try again later.';
    }
  }

  async function renderBooking(booking) {
    const card = document.createElement('div');
    card.className = 'booking-card card-style';
    card.innerHTML = `
      <div class="booking-info">
        <span class="booking-type-badge">${booking.listingType === 'car' ? 'Car' : 'House'}</span>
        <h3 class="listing-title">Loading listing…</h3>
        <p class="booking-meta">${formatDateRange(booking.startDate, booking.endDate)}</p>
        <p class="booking-total">${formatPrice(booking.totalPrice)}</p>
      </div>
      <div class="booking-status-block">
        ${renderStatusBadge(booking.status)}
      </div>
    `;
    list.appendChild(card);

    // Fill in the listing title once fetched (non-blocking — the
    // booking card itself renders immediately with what we already
    // have from Booking.js).
    const titleEl = card.querySelector('.listing-title');
    try {
      const listing = await fetchListing(booking.listingId, booking.listingType);
      titleEl.textContent = listing?.title || 'Listing';
    } catch {
      titleEl.textContent = 'Listing';
    }
  }

  async function fetchListing(listingId, listingType) {
    const endpoint = listingType === 'car' ? 'cars' : 'houses';
    const res = await fetch(`${API_BASE_URL}/api/${endpoint}/${listingId}`, {
      credentials: 'include',
    });
    if (!res.ok) return null;
    const data = await res.json();
    return data.house || data.car || data;
  }

  function renderStatusBadge(status) {
    if (status === 'cancelled') {
      return `<span class="status-badge cancelled">Cancelled</span>`;
    }

    if (status === 'confirmed') {
      return `
        <span class="status-badge confirmed">Confirmed</span>
        <p class="status-note">All set — see you soon!</p>
      `;
    }

    // status === 'pending' (or anything unexpected) — see Known Issue
    // #11 note at the top of this file for why this shows "Confirmed"
    // rather than "Pending".
    return `
      <span class="status-badge confirmed">Confirmed</span>
      <p class="status-note">We're finalizing a few last details — you'll hear from us if anything changes.</p>
    `;
  }

  function formatDateRange(startDate, endDate) {
    const opts = { month: 'short', day: 'numeric', year: 'numeric' };
    const start = new Date(startDate).toLocaleDateString(undefined, opts);
    const end = new Date(endDate).toLocaleDateString(undefined, opts);
    return `${start} – ${end}`;
  }

  function formatPrice(amount) {
    const value = Number(amount) || 0;
    return `$${value.toFixed(2)} total`;
  }

  function showLoginRequired() {
    loadingState.style.display = 'none';
    loginRequiredState.style.display = 'block';
  }

  }

  // Post-auto-approve model: pending should be rare now (only leftover
  // reviews from before the flip, or if moderation ever gets turned
  // back on) — those still bubble to the top since they need real
  // action. Everything else sorts newest-first, since this page's job
  // shifted from "clear the queue" to "spot-check what just went
  // live."
  function sortForSpotCheck(items) {
    return [...items].sort((a, b) => {
      const approvedDiff = Number(a.approved) - Number(b.approved);
      if (approvedDiff !== 0) return approvedDiff;
      const aTime = new Date(a.createdAt || 0).getTime();
      const bTime = new Date(b.createdAt || 0).getTime();
      return bTime - aTime;
    });
  }

  // Review.js confirms `rating` is the only field (Number, 1-5, no
  // "stars" alternate) — no fallback needed anymore.
  function getRating(review) {
    const num = Number(review.rating);
    if (!num || num < 1) return null;
    return Math.min(5, Math.round(num));
  }

  // Review.js confirms `comment` is the only field (String) — no
  // fallback needed anymore.
  function getComment(review) {
    return review.comment || null;
  }

  // Review.js confirms the ref field is named `userId`, and the
  // CONFIRMED public route (reviewController.js's getReviewsForListing)
  // populates it with just `name`. The ADMIN list endpoint this page
  // calls is a different, still-unconfirmed function — if it doesn't
  // populate the same way, `userId` arrives as a raw ObjectId string
  // instead of an object, which the fallback below still handles.
  function getReviewerLabel(review) {
    const candidate = review.userId;
    if (candidate && typeof candidate === 'object') {
      if (candidate.name) return candidate.name;
      if (candidate.email) return candidate.email;
    }
    if (typeof candidate === 'string') return `Reviewer ID: ${candidate}`;
    return 'Anonymous';
  }

  function getListingType(review) {
    const type = review.listingType;
    return (type === 'house' || type === 'car') ? type : null;
  }

  function getListingId(review) {
    return review.listingId || null;
  }

  function formatDate(value) {
    if (!value) return null;
    const d = new Date(value);
    if (isNaN(d.getTime())) return null;
    return d.toLocaleDateString(undefined, { year: 'numeric', month: 'short', day: 'numeric' });
  }

  // Same per-item title lookup pattern D4 (bookings.js) already uses.
  async function fetchListingTitle(review) {
    const type = getListingType(review);
    const id = getListingId(review);
    if (!type || !id) return 'Listing unavailable';

    const endpoint = type === 'house' ? 'houses' : 'cars';
    try {
      const res = await fetch(`${API_BASE_URL}/api/${endpoint}/${id}`, {
        method: 'GET',
        credentials: 'include'
      });
      if (!res.ok) return 'Listing unavailable';
      const data = await res.json();
      return data.title || 'Listing unavailable';
    } catch (err) {
      console.error(`Failed to fetch ${type} ${id} for review:`, err);
      return 'Listing unavailable';
    }
  }

  function buildCard(review, listingTitle) {
    const card = document.createElement('div');
    card.className = 'review-card';

    const info = document.createElement('div');
    info.className = 'review-info';

    const type = getListingType(review);
    const titleLine = document.createElement('div');
    titleLine.className = 'title-line';
    titleLine.textContent = listingTitle;
    if (type) {
      const typeTag = document.createElement('span');
      typeTag.className = 'type-tag';
      typeTag.textContent = type;
      titleLine.appendChild(typeTag);
    }
    info.appendChild(titleLine);

    const rating = getRating(review);
    if (rating) {
      const stars = document.createElement('div');
      stars.className = 'stars';
      stars.textContent = '★'.repeat(rating) + '☆'.repeat(5 - rating);
      info.appendChild(stars);
    }

    const comment = getComment(review);
    if (comment) {
      const commentEl = document.createElement('div');
      commentEl.className = 'comment';
      commentEl.textContent = comment;
      info.appendChild(commentEl);
    }

    const meta = document.createElement('div');
    meta.className = 'meta';
    const datePart = formatDate(review.createdAt);
    meta.textContent = [getReviewerLabel(review), datePart].filter(Boolean).join(' · ');
    info.appendChild(meta);

    const badge = document.createElement('span');
    badge.className = 'status-badge ' + (review.approved ? 'approved' : 'pending');
    badge.textContent = review.approved ? 'approved' : 'pending';

    const actions = document.createElement('div');
    actions.className = 'review-actions';

    const approveBtn = document.createElement('button');
    approveBtn.className = 'approve-btn';
    approveBtn.textContent = 'Approve';
    approveBtn.disabled = !!review.approved;

    const deleteBtn = document.createElement('button');
    deleteBtn.className = 'delete-btn';
    deleteBtn.textContent = 'Delete';

    approveBtn.addEventListener('click', () => approveReview(review._id, card));
    deleteBtn.addEventListener('click', () => deleteReview(review._id, card));

    actions.appendChild(approveBtn);
    actions.appendChild(deleteBtn);

    card.appendChild(info);
    card.appendChild(badge);
    card.appendChild(actions);

    return card;
  }

  async function approveReview(id, cardEl) {
    const buttons = cardEl.querySelectorAll('button');
    buttons.forEach((b) => (b.disabled = true));
    clearError();

    try {
      const res = await fetch(`${API_BASE_URL}/api/admin/reviews/${id}/approve`, {
        method: 'PUT',
        credentials: 'include'
      });

      if (res.status === 401 || res.status === 403) {
        window.location.href = 'login.html';
        return;
      }

      if (!res.ok) {
        showError('Something went wrong approving that review. Please try again.');
        buttons.forEach((b) => (b.disabled = false));
        return;
      }

      await loadAll();

    } catch (err) {
      console.error(`Failed to approve review ${id}:`, err);
      showError('Something went wrong approving that review. Please try again.');
      buttons.forEach((b) => (b.disabled = false));
    }
  }

  async function deleteReview(id, cardEl) {
    const confirmed = window.confirm('Delete this review permanently? This can\'t be undone.');
    if (!confirmed) return;

    const buttons = cardEl.querySelectorAll('button');
    buttons.forEach((b) => (b.disabled = true));
    clearError();

    try {
      const res = await fetch(`${API_BASE_URL}/api/admin/reviews/${id}`, {
        method: 'DELETE',
        credentials: 'include'
      });

      if (res.status === 401 || res.status === 403) {
        window.location.href = 'login.html';
        return;
      }

      if (!res.ok) {
        showError('Something went wrong deleting that review. Please try again.');
        buttons.forEach((b) => (b.disabled = false));
        return;
      }

      await loadAll();

    } catch (err) {
      console.error(`Failed to delete review ${id}:`, err);
      showError('Something went wrong deleting that review. Please try again.');
      buttons.forEach((b) => (b.disabled = false));
    }
  }

  async function loadAll() {
    try {
      const res = await fetch(`${API_BASE_URL}/api/admin/reviews`, {
        method: 'GET',
        credentials: 'include'
      });

      if (res.status === 401 || res.status === 403) {
        window.location.href = 'login.html';
        return;
      }

      if (!res.ok) {
        loadingMsg.style.display = 'none';
        showError('Something went wrong loading reviews. Please try again.');
        return;
      }

      const reviews = await res.json();
      loadingMsg.style.display = 'none';
      reviewsList.innerHTML = '';

      if (!reviews || reviews.length === 0) {
        emptyMsg.style.display = 'block';
        return;
      }
      emptyMsg.style.display = 'none';

      const sorted = sortForSpotCheck(reviews);
      const titles = await Promise.all(sorted.map(fetchListingTitle));
      sorted.forEach((review, i) => {
        reviewsList.appendChild(buildCard(review, titles[i]));
      });

    } catch (err) {
      console.error('Failed to load reviews:', err);
      loadingMsg.style.display = 'none';
      showError('Something went wrong loading reviews. Please try again.');
    }
  }

  logoutBtn.addEventListener('click', async function () {
    try {
      await fetch(`${API_BASE_URL}/api/auth/logout`, {
        method: 'POST',
        credentials: 'include'
      });
    } catch (err) {
      console.error('Logout request failed:', err);
    } finally {
      window.location.href = 'login.html';
    }
  });

  loadAll();
})();

// wishlist.js — C7: Wishlist page logic
//
// ASSUMPTIONS (wishlistRoutes.js / wishlistController.js content has
// never been read — see SAFRA-PROJECT-REFERENCE.txt Section 4, which
// only confirms "/api/wishlist/... - toggle + fetch wishlist" exists
// as a general description, not exact verbs/paths):
//
//   1. GET  ${API_BASE_URL}/api/wishlist
//      -> returns the current user's wishlist, already populated with
//         listing details, as an array of objects shaped like:
//         { listingId, listingType: "house" | "car", title,
//           location, pricePerNight / pricePerDay, photos: [] }
//      If the real response is instead just an array of bare IDs
//      (unpopulated), this code will need an extra fetch-per-item step
//      added in loadWishlist() below.
//
//   2. POST ${API_BASE_URL}/api/wishlist/toggle
//      body: { listingId, listingType }
//      -> adds if not present, removes if present (matches the
//         "toggle" naming in the route map). Used here for the
//         Remove button — calling toggle on an item already in the
//         wishlist should remove it.
//
//   3. Detail page links assume house-detail.html?houseId=<id> for
//      houses (this filename was never confirmed — C4 was built
//      without sharing the actual filename in chat — adjust below if
//      yours is named differently, e.g. house.html or listing.html)
//      and car-detail.html?carId=<id> for cars (confirmed, built as
//      C6 in this same chat).
//
//   4. Auth check uses GET ${API_BASE_URL}/api/auth/me (confirmed to
//      exist, added since Phase A) to detect whether the visitor is
//      logged in before attempting to load the wishlist.

(function () {
  const grid = document.getElementById('wishlist-grid');
  const loadingState = document.getElementById('loading-state');
  const loginRequiredState = document.getElementById('login-required-state');
  const emptyState = document.getElementById('empty-state');

  checkAuthThenLoad();

  async function checkAuthThenLoad() {
    try {
      const meRes = await fetch(`${API_BASE_URL}/api/auth/me`, {
        credentials: 'include',
      });

      if (meRes.status === 401) {
        showLoginRequired();
        return;
      }

      if (!meRes.ok) {
        showLoginRequired();
        return;
      }

      loadWishlist();
    } catch (err) {
      console.error('Auth check failed:', err);
      showLoginRequired();
    }
  }

  async function loadWishlist() {
    try {
      const res = await fetch(`${API_BASE_URL}/api/wishlist`, {
        credentials: 'include',
      });

      if (res.status === 401) {
        showLoginRequired();
        return;
      }

      if (!res.ok) {
        loadingState.textContent = 'Something went wrong loading your wishlist. Please try again later.';
        return;
      }

      const data = await res.json();
      const items = data.wishlist || data.items || data || [];

      loadingState.style.display = 'none';

      if (!items.length) {
        emptyState.style.display = 'block';
        return;
      }

      items.forEach(renderItem);
    } catch (err) {
      console.error('Failed to load wishlist:', err);
      loadingState.textContent = 'Could not reach the server. Please try again later.';
    }
  }

  function renderItem(item) {
    const listingType = item.listingType; // "house" | "car"
    const listingId = item.listingId || item._id;

    const detailUrl =
      listingType === 'car'
        ? `car-detail.html?carId=${encodeURIComponent(listingId)}`
        : `house-detail.html?houseId=${encodeURIComponent(listingId)}`;

    const price = Number(item.pricePerNight ?? item.pricePerDay) || 0;
    const priceLabel = listingType === 'car' ? `$${price.toFixed(2)} / day` : `$${price.toFixed(2)} / night`;

    const card = document.createElement('div');
    card.className = 'wishlist-item card-style';
    card.innerHTML = `
      <img src="${item.photos && item.photos[0] ? item.photos[0] : ''}" alt="${item.title || 'Listing photo'}" />
      <span class="item-type-badge">${listingType === 'car' ? 'Car' : 'House'}</span>
      <h3>${escapeHtml(item.title || 'Untitled listing')}</h3>
      <p class="location">${escapeHtml(item.location || '')}</p>
      <p class="price">${priceLabel}</p>
      <div class="item-actions">
        <a href="${detailUrl}" class="btn">View</a>
        <button type="button" class="btn-remove">Remove</button>
      </div>
    `;

    card.querySelector('.btn-remove').addEventListener('click', () => {
      removeFromWishlist(listingId, listingType, card);
    });

    grid.appendChild(card);
  }

  async function removeFromWishlist(listingId, listingType, cardEl) {
    const removeBtn = cardEl.querySelector('.btn-remove');
    removeBtn.disabled = true;
    removeBtn.textContent = 'Removing…';

    try {
      const res = await fetch(`${API_BASE_URL}/api/wishlist/toggle`, {
        method: 'POST',
        credentials: 'include',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ listingId, listingType }),
      });

      if (!res.ok) {
        removeBtn.disabled = false;
        removeBtn.textContent = 'Remove';
        alert('Could not remove this item. Please try again.');
        return;
      }

      cardEl.remove();

      if (!grid.children.length) {
        emptyState.style.display = 'block';
      }
    } catch (err) {
      console.error('Failed to remove wishlist item:', err);
      removeBtn.disabled = false;
      removeBtn.textContent = 'Remove';
      alert('Could not reach the server. Please try again.');
    }
  }

  function escapeHtml(str) {
    const div = document.createElement('div');
    div.textContent = str;
    return div.innerHTML;
  }

  function showLoginRequired() {
    loadingState.style.display = 'none';
    loginRequiredState.style.display = 'block';
  }
})();
