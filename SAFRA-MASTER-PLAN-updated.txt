Routes (routes/featured.js) â€” final version, houses AND cars both

@'
const express = require('express');
const router = express.Router();
const Featured = require('../models/Featured');
const House = require('../models/House');
const Car = require('../models/Car');

// Admin: candidates to pick from
router.get('/candidates', async (req, res) => {
  try {
    const houses = await House.find({}, 'title location pricePerNight').lean();
    const cars = await Car.find({}, 'title location pricePerDay').lean();
    res.json({ houses, cars });
  } catch (err) {
    res.status(500).json({ error: err.message });
  }
});

// Public: get the current featured picks for the homepage
router.get('/', async (req, res) => {
  try {
    const entries = await Featured.find().lean();
    const houseIds = entries.filter(e => e.itemType === 'House').map(e => e.refId);
    const carIds = entries.filter(e => e.itemType === 'Car').map(e => e.refId);

    const houses = await House.find({ _id: { $in: houseIds } }).limit(2);
    const cars = await Car.find({ _id: { $in: carIds } }).limit(1);

    res.json({ houses, car: cars[0] || null });
  } catch (err) {
    res.status(500).json({ error: err.message });
  }
});

// Admin: set the featured picks (replaces current selection)
// body: { houseIds: [...], carIds: [...] }
router.post('/', async (req, res) => {
  try {
    const { houseIds = [], carIds = [] } = req.body;
    await Featured.deleteMany({});

    const docs = [];
    houseIds.slice(0, 2).forEach(id => docs.push({ itemType: 'House', refId: id }));
    carIds.slice(0, 1).forEach(id => docs.push({ itemType: 'Car', refId: id }));

    await Featured.insertMany(docs);
    res.json({ success: true });
  } catch (err) {
    res.status(500).json({ error: err.message });
  }
});

module.exports = router;

--- 4c. Frontend: homepage renderer (frontend/js/featured-listings.js) ---
Fetches GET /api/featured and renders into #featuredGrid on index.html.
Houses are always pushed into the cards array before the car, so the
homepage order is guaranteed: 2 houses first, then 1 car.
@'
async function loadFeatured() {
  const grid = document.getElementById("featuredGrid");
  if (!grid) return;

  try {
    const res = await fetch(`${API_BASE_URL}/api/featured`);
    const data = await res.json();

    const cards = [];

    (data.houses || []).forEach(h => {
      cards.push(`
        <a href="houses.html" class="listing-card">
          <div class="listing-thumb"><img src="${h.photos[0]}" alt="${h.title}"></div>
          <div class="listing-body">
            <div class="listing-title">${h.title}</div>
            <div class="listing-location">${h.location}</div>
            <div class="listing-footer-row">
              <div class="listing-price">$${h.pricePerNight}<span>per night</span></div>
              <span class="listing-cta">View Details</span>
            </div>
          </div>
        </a>
      `);
    });

    if (data.car) {
      const c = data.car;
      cards.push(`
        <a href="cars.html" class="listing-card">
          <div class="listing-thumb"><img src="${c.photos[0]}" alt="${c.title}"></div>
          <div class="listing-body">
            <div class="listing-title">${c.title}</div>
            <div class="listing-location">${c.location}</div>
            <div class="listing-footer-row">
              <div class="listing-price">$${c.pricePerDay}<span>per day</span></div>
              <span class="listing-cta">View Details</span>
            </div>
          </div>
        </a>
      `);
    }

    grid.classList.remove("featured-loading");
    grid.innerHTML = cards.join("") || `<div class="listing-empty">No featured listings yet.</div>`;
  } catch (err) {
    console.error("Failed to load featured listings:", err);
    grid.innerHTML = `<div class="listing-error">Could not load featured listings.</div>`;
  }
}
loadFeatured();
